```python indicators.py
"""
Financial Technical Indicators Module.
Implement the functions below adhering to production standards and numerical robustness.
"""

import numpy as np
import pandas as pd


def compute_rolling_volatility(
    prices: pd.Series,
    window: int = 20,
    annualized: bool = True,
    trading_days: int = 252,
) -> pd.Series:
    """
    Computes rolling realized volatility of percentage returns.
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    # Check for non-positive prices
    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    # If length <= window, return all NaNs with identical index
    if len(prices) <= window:
        return pd.Series(np.nan, index=prices.index)

    # Compute percentage returns: prices.pct_change() yields NaN at index 0
    returns = prices.pct_change()

    # Pandas rolling default uses closed='right' and min_periods=window
    # To get window valid returns where the first return is at index 1,
    # we need window observations of returns.
    # The first valid return is at index 1. A rolling window of size W on returns
    # will complete at index W (since indices 1 to W inclusive give W returns).
    # Specifically, rolling(window=window).std(ddof=1) on returns:
    rolling_std = returns.rolling(window=window, min_periods=window).std(ddof=1)

    # Handle zero variance / flat price periods gracefully: if prices are constant over
    # a completed window, rolling_std might be NaN due to std of zeros or 0.0.
    # Actually, pandas std of constant series is 0.0 (if min_periods met).
    # However, sometimes floating point or edge cases might give NaN or 0.0.
    # We ensure that where returns are available and constant (std is 0.0 or NaN due to zero variance),
    # it evaluates to 0.0 once the window is completed.
    # Wait, pandas rolling std of a constant series (where all returns are 0.0) is 0.0.
    # Let's explicitly fill NaN with 0.0 where the rolling window is fully populated by constant prices.
    # A rolling window is populated if rolling_std is not NaN. If standard deviation calculation
    # results in NaN because all values are identical, pandas returns 0.0 for std.
    # But just to be robust:
    rolling_std = rolling_std.fillna(0.0)

    # Re-apply NaN for warmup bars where min_periods wasn't met (indices 0 through window - 1)
    # Actually, rolling(window=window, min_periods=window) naturally produces NaN for indices 0 to window - 1.
    # But wait, what about index 0 where returns is NaN? rolling with min_periods=window will be NaN for 0 to window-1.
    # Let's verify: returns has NaN at index 0. For window=20, rolling(20).std() needs 20 non-NaN values.
    # The first 20 returns are at indices 1 through 20 (total 20 returns). So index 20 is the first valid value.
    # This corresponds exactly to index `window` (since index is 0-based, index `window` is the (window+1)-th bar).
    # Indices 0 through window-1 will be NaN.

    if annualized:
        volatility = rolling_std * np.sqrt(trading_days)
    else:
        volatility = rolling_std

    # Ensure indices 0 through window-1 are strictly NaN (in case fillna(0.0) affected anything, though fillna(0.0)
    # on NaN from warmup would turn warmup into 0.0! That would be a bug!)
    # Ah! Good catch! `rolling_std.fillna(0.0)` would turn the warmup NaN values into 0.0!
    # We must only fillna(0.0) where the window is actually completed (i.e. where returns are valid/populated).
    
    # Let's write this more precisely:
    # A rolling window has completed if the number of non-null returns in the window equals `window`.
    # Alternatively, we can check where `returns.rolling(window=window, min_periods=window).count() == window`.
    completed_mask = returns.rolling(window=window, min_periods=window).count() == window

    # Where completed and rolling_std is NaN (e.g. zero variance where std might be NaN or 0), set to 0.0.
    # Pandas std on constant values is 0.0, but let's be explicit.
    rolling_std = rolling_std.where(completed_mask, np.nan)
    rolling_std = rolling_std.mask(completed_mask & rolling_std.isna(), 0.0)

    if annualized:
        volatility = rolling_std * np.sqrt(trading_days)
    else:
        volatility = rolling_std

    return volatility


def compute_exponential_momentum(
    prices: pd.Series,
    span: int = 14,
) -> pd.Series:
    """
    Computes normalized exponential moving average distance:
    momentum = (prices - EMA) / EMA
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    # Check for non-positive prices
    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    ema = prices.ewm(span=span, adjust=False).mean()
    momentum = (prices - ema) / ema

    return momentum
```

```markdown NOTES.md
## Technical Assumptions

### 1. Rolling Window Closure Convention (`closed='right'`)
- All rolling window calculations utilize the standard pandas default of `closed='right'`, meaning that timestamp $T$ incorporates the current observation at $T$ along with the preceding $window - 1$ historical observations.

### 2. Warmup & `NaN` Policy
- Percentage returns (`prices.pct_change()`) inherently produce `NaN` at index 0.
- Computing a rolling statistic over a window of size $W$ requires $W$ valid return values, which necessitates $W + 1$ price observations.
- Consequently, the first valid (non-`NaN`) rolling volatility observation occurs at index `window` (corresponding to the $(window + 1)$-th price bar).
- All initial price bars from index $0$ through $window - 1$ evaluate strictly to `NaN`.
- If the length of the price series is less than or equal to $window$ (`len(prices) <= window`), the entire output series consists of `NaN`s with an identical index and length to `prices`.
- Zero variance or flat price periods over a completed window evaluate gracefully to `0.0` volatility (while warmup bars remain `NaN`).

### 3. Annualization Basis
- Realized volatility annualization assumes a standard equity trading calendar year consisting of `trading_days=252`, scaling the periodic standard deviation by $\sqrt{252}$ when `annualized=True`.
```
