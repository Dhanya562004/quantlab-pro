```python indicators.py
"""
Financial Technical Indicators Module.
Implement the functions below adhering to production standards and numerical robustness.
"""

import numpy as np
import pandas as pd


def compute_rolling_volatility(
    prices: pd.Series,
    window: int = 20,
    annualized: bool = True,
    trading_days: int = 252,
) -> pd.Series:
    """
    Computes rolling realized volatility of percentage returns (prices.pct_change()).
    Uses sample standard deviation with Bessel's correction (ddof=1).
    If annualized=True, multiplies rolling standard deviation by sqrt(trading_days).
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    # Percentage returns: len(prices) elements, index 0 is NaN
    returns = prices.pct_change()

    # If len(prices) <= window, we cannot form a full window of returns.
    # Specifically, a window of W returns requires W+1 prices.
    # Therefore, if len(prices) <= window, return all NaNs.
    if len(prices) <= window:
        return pd.Series(np.nan, index=prices.index, dtype=float)

    # pandas rolling standard deviation with ddof=1 (default for pandas rolling.std)
    # closed='right' is standard pandas rolling default
    rolling_std = returns.rolling(window=window, closed='right').std(ddof=1)

    if annualized:
        rolling_vol = rolling_std * np.sqrt(trading_days)
    else:
        rolling_vol = rolling_std

    # Ensure zero variance / flat price periods evaluate to 0.0 for completed windows
    # (where rolling_std evaluates to 0.0 instead of NaN).
    # Warmup bars (< window) must still evaluate to NaN.
    # rolling.std() on constant returns already evaluates to 0.0 in pandas, but let's be explicitly robust.
    # Note: indices 0 to window are NaN anyway due to return calculation and window warmup.

    return rolling_vol


def compute_exponential_momentum(
    prices: pd.Series,
    span: int = 14,
) -> pd.Series:
    """
    Computes normalized exponential moving average distance:
    momentum = (prices - EMA) / EMA
    where EMA = prices.ewm(span=span, adjust=False).mean()
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    ema = prices.ewm(span=span, adjust=False).mean()
    momentum = (prices - ema) / ema

    return momentum
```

```markdown NOTES.md
## Technical Assumptions

This document outlines the technical assumptions and boundary conventions implemented in the `indicators.py` module for financial technical indicators.

### 1. Rolling Window Closure Convention (`closed='right'`)
- Rolling calculations (such as rolling standard deviation of returns) follow the standard pandas rolling behavior with `closed='right'`. 
- This means that for timestamp $T$, the rolling window includes observation $T$ and the preceding $window - 1$ observations (inclusive of $T$).

### 2. Warmup & `NaN` Policy
- **Percentage Returns (`prices.pct_change()`)**: Generates a `NaN` at index 0 because a return requires a prior price bar.
- **Window Requirement**: Computing a rolling volatility window of size $W$ over percentage returns requires $W$ valid return values. Since index 0 is `NaN`, $W$ valid returns require $W + 1$ price bars (indices $0$ through $W$).
- **Output Indexing**: 
  - The first valid (non-`NaN`) rolling volatility observation occurs at index `window` (the $(window + 1)$-th price bar).
  - All initial price bars from index $0$ through $window - 1$ evaluate to `NaN`.
  - If `len(prices) <= window`, the function returns a `pd.Series` of all `NaN` values matching the input index and length.

### 3. Annualization Basis Factor ($\sqrt{252}$)
- When `annualized=True`, realized rolling volatility (sample standard deviation with Bessel's correction, `ddof=1`) is multiplied by the square root of the standard equity trading year basis, $\sqrt{252}$ (`trading_days=252`).

### 4. Zero Variance and Flat Price Handling
- If price series contain constant / flat values over a completed window ($\ge window$), percentage returns are $0.0$, and the sample standard deviation evaluates to `0.0`. Rolling volatility correctly yields `0.0` rather than `NaN` or unhandled division errors. Warmup bars ($< window$) remain `NaN`.

### 5. Input Validation
- Both indicator functions validate that all elements in `prices` are strictly positive ($> 0$). If any non-positive value ($\le 0$) is encountered, a `ValueError("Prices must be strictly positive")` is raised immediately.
```
