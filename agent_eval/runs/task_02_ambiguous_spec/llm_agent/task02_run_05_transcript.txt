```python indicators.py
"""
Financial Technical Indicators Module.
Implement the functions below adhering to production standards and numerical robustness.
"""

import numpy as np
import pandas as pd


def compute_rolling_volatility(
    prices: pd.Series,
    window: int = 20,
    annualized: bool = True,
    trading_days: int = 252,
) -> pd.Series:
    """
    Computes rolling realized volatility of percentage returns (prices.pct_change()).
    Uses sample standard deviation with Bessel's correction (ddof=1).
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    # Compute percentage returns: pct_change() results in NaN at index 0.
    returns = prices.pct_change()

    # Rolling standard deviation with ddof=1 (sample standard deviation).
    # Pandas rolling by default uses closed='right' and requires min_periods=window.
    rolling_std = returns.rolling(window=window, min_periods=window).std(ddof=1)

    if annualized:
        vol = rolling_std * np.sqrt(trading_days)
    else:
        vol = rolling_std

    # Ensure zero variance / flat price periods evaluate to 0.0 for completed windows.
    # Where rolling std is 0.0 (or very close to floating point zero), set to 0.0.
    # Note: NaN values from warmup should remain NaN.
    vol = vol.where(vol.isna() | (vol != 0), 0.0)

    # Ensure output index matches prices and is a Series
    vol.index = prices.index
    return vol


def compute_exponential_momentum(
    prices: pd.Series,
    span: int = 14,
) -> pd.Series:
    """
    Computes normalized exponential moving average distance:
    momentum = (prices - EMA) / EMA
    where EMA = prices.ewm(span=span, adjust=False).mean()
    """
    if not isinstance(prices, pd.Series):
        prices = pd.Series(prices)

    if (prices <= 0).any():
        raise ValueError("Prices must be strictly positive")

    ema = prices.ewm(span=span, adjust=False).mean()
    momentum = (prices - ema) / ema
    momentum.index = prices.index
    return momentum
```

```markdown NOTES.md
# Technical Notes

## Technical Assumptions

### 1. Rolling Window Closure Convention
- **Convention**: `closed='right'` (standard pandas rolling behavior).
- **Explanation**: The rolling window at timestamp $T$ includes the return observation at $T$ along with the preceding $window - 1$ return observations.

### 2. Warmup and NaN Policy
- **Convention**: Strict warmup preservation yielding `NaN` for all initial bars before a full window of returns is available.
- **Explanation**: 
  - Percentage returns (`prices.pct_change()`) inherently produce `NaN` at index 0.
  - Computing a rolling volatility of window $W$ requires $W$ valid return values, which necessitates $W + 1$ price bars.
  - Consequently, the first valid rolling volatility observation appears at index `window` (the $(window + 1)$-th price bar).
  - The initial $window$ price bars (indices $0$ through $window - 1$) evaluate to `NaN`.
  - If `len(prices) <= window`, the function returns a `pd.Series` of all `NaN` values with identical index and length to `prices`.

### 3. Annualization Basis Factor
- **Convention**: $\sqrt{252}$ trading days (`trading_days=252`).
- **Explanation**: Standard equity trading year convention where daily volatility is annualized by multiplying by the square root of 252.

### 4. Zero Variance / Flat Price Handling
- **Convention**: Flat price periods over completed windows evaluate to exact `0.0`.
- **Explanation**: If prices are constant over a completed window ($\ge window$), the return standard deviation is $0.0$ rather than `NaN` or unhandled division artifacts. Warmup bars ($< window$) remain `NaN`.
```
